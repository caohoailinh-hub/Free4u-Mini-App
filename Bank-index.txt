<!-- Script xử lý sự kiện Tab ủy quyền (Event Delegation) và phân trang / bảng -->
<script>
    document.addEventListener("DOMContentLoaded", function() {
        const btnBatchPost = document.getElementById("btnBatchPost");

        if (btnBatchPost && !btnBatchPost.dataset.bound) {
            btnBatchPost.dataset.bound = "true";
            btnBatchPost.addEventListener("click", function() {
                const checkedBoxes = document.querySelectorAll(".row-checkbox:checked");
                
                if (checkedBoxes.length === 0) {
                    alert("Vui lòng chọn ít nhất một giao dịch để ghi sổ!");
                    return;
                }

                const selectedIds = Array.from(checkedBoxes).map(cb => cb.value);

                if (!confirm(`Bạn có chắc chắn muốn ghi sổ ${selectedIds.length} giao dịch đã chọn không?`)) {
                    return;
                }

                fetch('/Bank/BatchPost', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify(selectedIds)
                })
                .then(response => response.json())
                .then(data => {
                    if (data.success) {
                        alert(data.message || "Ghi sổ thành công!");
                        location.reload();
                    } else {
                        alert("Lỗi: " + (data.message || "Không thể thực hiện ghi sổ."));
                    }
                })
                .catch(error => {
                    console.error("Lỗi kết nối:", error);
                    alert("Đã xảy ra lỗi trong quá trình kết nối đến máy chủ.");
                });
            });
        }
    });

    // Dùng Event Delegation để bắt sự kiện click tab
    const bankRoot = document.getElementById("bankTab")?.closest(".container-fluid");

    if (bankRoot && !bankRoot.dataset.bankEventsBound) {
        bankRoot.dataset.bankEventsBound = "true";

        bankRoot.addEventListener("click", function(e) {
            const tabBtn = e.target.closest(".tab-filter");
            if (!tabBtn || !bankRoot.contains(tabBtn)) return;

            bankRoot.querySelectorAll(".tab-filter").forEach(b => {
                b.classList.remove(
                    "active", "fw-semibold", "text-success",
                    "border-bottom", "border-success", "border-2"
                );
                b.classList.add("text-dark");
            });

            tabBtn.classList.add(
                "active", "fw-semibold", "text-success",
                "border-bottom", "border-success", "border-2"
            );
            tabBtn.classList.remove("text-dark");

            const filter = tabBtn.getAttribute("data-filter");

            if (typeof window.applyBankFilter === "function") {
                window.applyBankFilter(filter);
            }
        });
    }

    function initBankTableModule() {
        const table = document.querySelector(".table-sticky");
        if (!table) return;

        const tbody = table.querySelector("tbody");
        let rows = Array.from(tbody.querySelectorAll("tr:not(.no-data)"));
        const totalRecordsEl = document.getElementById("total-records");
        const footerTotalAmountEl = document.getElementById("footer-total-amount");

        const infoSpan = document.getElementById("pagination-info");
        const btnFirst = document.getElementById("btn-first");
        const btnPrev = document.getElementById("btn-prev");
        const btnNext = document.getElementById("btn-next");
        const btnLast = document.getElementById("btn-last");
        const pageSizeSelect = document.getElementById("pageSizeSelect");

        const toolbarThu = document.getElementById("toolbar-thu");
        const toolbarChi = document.getElementById("toolbar-chi");
        const selectAllCheckbox = document.getElementById("selectAll");

        let currentPage = 1;
        let pageSize = parseInt(pageSizeSelect ? pageSizeSelect.value : 20);
        let currentFilter = "all";

        // --- XỬ LÝ HIỆN/ẨN CỤM NÚT TÁC VỤ HÀNG LOẠT VÀ NÚT THÔNG THƯỜNG KHI TICK CHỌN ---
        const batchToolbar = document.getElementById("batchActionToolbar");
        const defaultToolbar = document.getElementById("defaultToolbarActions");
        const selectedCountEl = document.getElementById("selected-count");

        function updateBatchToolbarState() {
            let visibleRowCheckboxes = rows
                .filter(r => r.style.display !== "none")
                .map(r => r.querySelector(".row-checkbox"))
                .filter(cb => cb !== null);

            let checkedCount = visibleRowCheckboxes.filter(cb => cb.checked).length;

            if (checkedCount > 0) {
                // Hiển thị cụm nút hàng loạt, ẩn cụm nút thông thường bên phải
                batchToolbar.classList.remove("d-none");
                batchToolbar.classList.add("d-flex");
                
                if (defaultToolbar) {
                    defaultToolbar.classList.remove("d-flex");
                    defaultToolbar.classList.add("d-none");
                }

                if (selectedCountEl) selectedCountEl.textContent = checkedCount;
            } else {
                // Ẩn cụm nút hàng loạt, hiển thị lại cụm nút thông thường bên phải
                batchToolbar.classList.remove("d-flex");
                batchToolbar.classList.add("d-none");

                if (defaultToolbar) {
                    defaultToolbar.classList.remove("d-none");
                    defaultToolbar.classList.add("d-flex");
                }
            }

            // Đồng bộ trạng thái checkbox "Chọn tất cả"
            if (selectAllCheckbox) {
                selectAllCheckbox.checked = (visibleRowCheckboxes.length > 0 && checkedCount === visibleRowCheckboxes.length);
            }
        }

        if (tbody && !tbody.dataset.batchBound) {
            tbody.dataset.batchBound = "true";
            tbody.addEventListener("change", function(e) {
                if (e.target && e.target.classList.contains("row-checkbox")) {
                    updateBatchToolbarState();
                }
            });
        }

        if (selectAllCheckbox && !selectAllCheckbox.dataset.batchBound) {
            selectAllCheckbox.dataset.batchBound = "true";
            selectAllCheckbox.addEventListener("change", function() {
                let filteredRows = rows.filter(r => currentFilter === "all" || r.getAttribute("data-type") === currentFilter);
                filteredRows.forEach(row => {
                    const cb = row.querySelector(".row-checkbox");
                    if (cb && row.style.display !== "none") {
                        cb.checked = selectAllCheckbox.checked;
                    }
                });
                updateBatchToolbarState();
            });
        }

        window.applyBankFilter = function(filterVal) {
            currentFilter = filterVal;
            currentPage = 1;

            if (toolbarThu && toolbarChi) {
                if (currentFilter === "thu") {
                    toolbarThu.classList.remove("d-none");
                    toolbarChi.classList.add("d-none");
                } else if (currentFilter === "chi") {
                    toolbarThu.classList.add("d-none");
                    toolbarChi.classList.remove("d-none");
                } else {
                    toolbarThu.classList.remove("d-none");
                    toolbarChi.classList.remove("d-none");
                }
            }
            renderTable();
        };

        function renderTable() {
            let filteredRows = rows.filter(row => {
                if (currentFilter === "all") return true;
                return row.getAttribute("data-type") === currentFilter;
            });

            const totalFiltered = filteredRows.length;
            if (totalRecordsEl) totalRecordsEl.textContent = totalFiltered;

            let sumAmount = 0;
            filteredRows.forEach(row => {
                const amount = parseFloat(row.getAttribute("data-amount")) || 0;
                sumAmount += amount;
            });
            if (footerTotalAmountEl) {
                footerTotalAmountEl.textContent = sumAmount.toLocaleString('en-US');
            }

            const totalPages = Math.ceil(totalFiltered / pageSize) || 1;
            if (currentPage > totalPages) currentPage = totalPages;
            if (currentPage < 1) currentPage = 1;

            const startIdx = (currentPage - 1) * pageSize;
            const endIdx = Math.min(startIdx + pageSize, totalFiltered);

            rows.forEach(row => row.style.display = "none");
            for (let i = startIdx; i < endIdx; i++) {
                if (filteredRows[i]) {
                    filteredRows[i].style.display = "";
                }
            }

            if (infoSpan) {
                infoSpan.textContent = totalFiltered === 0 ? "0 - 0" : `${startIdx + 1} - ${endIdx}`;
            }

            if (btnFirst) btnFirst.disabled = (currentPage === 1);
            if (btnPrev) btnPrev.disabled = (currentPage === 1);
            if (btnNext) btnNext.disabled = (currentPage >= totalPages || totalPages === 0);
            if (btnLast) btnLast.disabled = (currentPage >= totalPages || totalPages === 0);
        }

        if (pageSizeSelect && !pageSizeSelect.dataset.bound) {
            pageSizeSelect.dataset.bound = "true";
            pageSizeSelect.addEventListener("change", function() {
                pageSize = parseInt(this.value);
                currentPage = 1;
                renderTable();
            });
        }

        if (btnFirst && !btnFirst.dataset.bound) {
            btnFirst.dataset.bound = "true";
            btnFirst.addEventListener("click", () => { if (currentPage > 1) { currentPage = 1; renderTable(); } });
            btnPrev.addEventListener("click", () => { if (currentPage > 1) { currentPage--; renderTable(); } });
            btnNext.addEventListener("click", () => {
                let filteredRows = rows.filter(r => currentFilter === "all" || r.getAttribute("data-type") === currentFilter);
                let totalPages = Math.ceil(filteredRows.length / pageSize) || 1;
                if (currentPage < totalPages) { currentPage++; renderTable(); }
            });
            btnLast.addEventListener("click", () => {
                let filteredRows = rows.filter(r => currentFilter === "all" || r.getAttribute("data-type") === currentFilter);
                let totalPages = Math.ceil(filteredRows.length / pageSize) || 1;
                if (currentPage < totalPages) { currentPage = totalPages; renderTable(); }
            });
        }

        renderTable();
    }

    initBankTableModule();
</script>

<!-- Script điều khiển cụm nút Xem - Sửa - Xóa nổi ở mép phải bảng -->
<script>
    document.addEventListener("DOMContentLoaded", function() {
        const actionPanel = document.getElementById("globalRowActions");
        const tableContainer = document.querySelector(".table-responsive");

        if (!actionPanel || !tableContainer) return;

        const btnView = document.getElementById("btnViewRow");
        const btnEdit = document.getElementById("btnEditRow");
        const btnDelete = document.getElementById("btnDeleteRow");

        let currentRow = null;
        let hideTimer = null;

        function hideActions() {
            window.clearTimeout(hideTimer);
            currentRow = null;
            actionPanel.classList.remove("show");
        }

        function updateActionPosition() {
            if (!currentRow || !currentRow.isConnected ||
                currentRow.style.display === "none") {
                hideActions();
                return;
            }

            const rowRect = currentRow.getBoundingClientRect();
            const containerRect = tableContainer.getBoundingClientRect();

            const visibleTop = Math.max(containerRect.top, 0);
            const visibleBottom = Math.min(containerRect.bottom, window.innerHeight);

            if (rowRect.bottom <= visibleTop || rowRect.top >= visibleBottom) {
                actionPanel.classList.remove("show");
                return;
            }

            actionPanel.classList.add("show");

            const panelWidth = actionPanel.offsetWidth;
            const panelHeight = actionPanel.offsetHeight;

            let left = Math.min(
                containerRect.right - panelWidth - 8,
                window.innerWidth - panelWidth - 8
            );
            left = Math.max(8, left);

            let top = rowRect.top + (rowRect.height - panelHeight) / 2;
            top = Math.max(5, Math.min(top, window.innerHeight - panelHeight - 5));

            actionPanel.style.left = left + "px";
            actionPanel.style.top = top + "px";
        }

        function scheduleHide() {
            window.clearTimeout(hideTimer);
            hideTimer = window.setTimeout(function() {
                if (!actionPanel.matches(":hover") &&
                    !(currentRow && currentRow.matches(":hover"))) {
                    hideActions();
                }
            }, 100);
        }

        tableContainer.addEventListener("mouseover", function(event) {
            const row = event.target.closest("tr[data-transaction-id]");
            if (!row || !tableContainer.contains(row)) return;

            window.clearTimeout(hideTimer);
            currentRow = row;
            updateActionPosition();
        });

        tableContainer.addEventListener("mouseout", function(event) {
            const row = event.target.closest("tr[data-transaction-id]");
            if (!row) return;

            const nextTarget = event.relatedTarget;
            if (nextTarget && (row.contains(nextTarget) || actionPanel.contains(nextTarget))) {
                return;
            }

            scheduleHide();
        });

        actionPanel.addEventListener("mouseenter", function() {
            window.clearTimeout(hideTimer);
        });

        actionPanel.addEventListener("mouseleave", scheduleHide);

        tableContainer.addEventListener("scroll", updateActionPosition, { passive: true });
        window.addEventListener("scroll", updateActionPosition, { passive: true });
        window.addEventListener("resize", updateActionPosition);

        function runRowAction(actionName) {
            if (!currentRow) return;

            const transactionId = currentRow.getAttribute("data-transaction-id");
            hideActions();

            const actionFunction = window[actionName];
            if (typeof actionFunction === "function") {
                actionFunction(transactionId);
            } else {
                console.error(
                    "Chưa khai báo hàm " + actionName +
                    "(id). Hãy kiểm tra hàm xử lý Xem/Sửa/Xóa trong View hoặc file JavaScript."
                );
            }
        }

        if (btnView && !btnView.dataset.bound) {
            btnView.dataset.bound = "true";
            btnView.addEventListener("click", function() {
                runRowAction("viewRow");
            });
        }

        if (btnEdit && !btnEdit.dataset.bound) {
            btnEdit.dataset.bound = "true";
            btnEdit.addEventListener("click", function() {
                runRowAction("editRow");
            });
        }

        if (btnDelete && !btnDelete.dataset.bound) {
            btnDelete.dataset.bound = "true";
            btnDelete.addEventListener("click", function() {
                runRowAction("deleteRow");
            });
        }
    });
</script>
